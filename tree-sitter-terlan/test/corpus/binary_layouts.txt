module binary.Tooling.

pub packet(port: Int, scalar: Int, body: Dynamic): Dynamic ->
  Binary[big] { port: UInt[16], scalar: Utf8, body: Rest }.

pub decode(value: Dynamic): Int ->
  case value {
    Binary[little] { port: UInt[16], delta: IntBits[8] } -> port + delta;
    _ -> 0
  }.

pub decode_head(Binary[big] { port: UInt[16], body: Rest }): Int ->
  port.

pub decode_lambda(value: Dynamic): Int ->
  let decode = ((Binary[big] { port: UInt[16], body: Rest }) -> port);
  decode(value).

pub decode_let(value: Dynamic): Int ->
  let Binary[big] { prefix: Bytes[2], flags: Bits[3], body: Rest } = value;
  flags.bit_length().

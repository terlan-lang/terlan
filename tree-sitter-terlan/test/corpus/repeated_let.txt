module app.GroupedLetElse.

pub resolve(first: Result[Int, String], second: Result[Int, String]): Result[Int, String] ->
  if {
      Ok(left) <- first;
      Ok(right) <- second
  } else {
      Err(reason) -> Err(reason)
  };
  Ok(left + right).
